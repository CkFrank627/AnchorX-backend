function renderMessages(notifications) {
  tabContent.innerHTML = '';

    tabContent.innerHTML = '<p class="no-message-text">暂无消息</p>';
    return;
  }

  const pageSize = 8;

  const totalPages = Math.ceil(notifications.length / pageSize);

  // ✅ 渲染当前页函数
  function renderPage(page) {
    messageSection.innerHTML = '';
    const start = (page - 1) * pageSize;
    const end = start + pageSize;
    const currentItems = notifications.slice(start, end);

    currentItems.forEach(notification => {
      const messageItem = document.createElement('div');
      messageItem.className = 'message-item';
      messageItem.style.cursor = 'pointer';
      if (!notification.read) messageItem.classList.add('unread');

      const senderName = notification.sender?.username || '未知用户';
      const messageTime = new Date(notification.createdAt).toLocaleString();
      let messageTitle = notification.message || '';
      if (!messageTitle.includes(senderName)) {
        messageTitle = `${senderName} ${messageTitle}`;
      }

      messageItem.innerHTML = `
        <div class="message-info">
          <p class="message-title">${messageTitle}</p>
          <p class="message-time">${messageTime}</p>

           // 🔥 新版：点击消息 = 标记已读 + 跳转到对应帖子/作品

            method: 'POST',
            headers: { 'Authorization': `Bearer ${token}` }

        // 如果当前是 "t" (繁体), direction 为 "t2s" (繁转简)
        const direction = currentLang === "s" ? "s2t" : "t2s"; 

        function getTextNodes(node) {
            let textNodes = [];
            for (let child of node.childNodes) {
                if (child.nodeType === Node.TEXT_NODE && child.textContent.trim()) {
                    // 忽略 <style> 和 <script> 里的文本节点
                    if (node.tagName !== "STYLE" && node.tagName !== "SCRIPT") {
                        textNodes.push(child);
                    }
                } else if (child.nodeType === Node.ELEMENT_NODE) {
                    // 同样跳过不需要转换的标签
                    if (!["STYLE", "SCRIPT", "SVG"].includes(child.tagName)) {
                        textNodes = textNodes.concat(getTextNodes(child));
                    }
                }
            }
            return textNodes;
        }


        const textNodes = getTextNodes(document.body);
        const fullText = textNodes.map(n => n.textContent).join("\n");

                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ text: fullText, direction })
            });

            const data = await response.json();
            const convertedText = data.convertedText.split("\n");

            textNodes.forEach((node, i) => {
                node.textContent = convertedText[i] || "";
            });

            // 更新语言状态和按钮文本

        } catch (err) {

            alert("繁简转换失败，请稍后重试。");
        } finally {

            progressOverlay.classList.add('hidden');
        }
    });
});

// 根据不同结构的通知对象，解析出应该跳转的 URL
function getNotificationTargetUrl(notification) {
  // 调试时可以先看一下后端到底给了什么字段
  console.log('📩 解析通知跳转链接：', notification);

  // 1️⃣ 后端如果直接给了 URL，优先使用
  if (notification.url) return notification.url;
  if (notification.targetUrl) return notification.targetUrl;
  if (notification.link) return notification.link;

  // 2️⃣ 可选：构造评论锚点（如果你阅读页/讨论区支持 #reply-xxx）
  let anchor = '';
  if (notification.comment) {
    anchor = `#reply-${notification.comment}`;
  } else if (notification.likedComment) {
    anchor = `#reply-${notification.likedComment}`;
  }

  // 3️⃣ 【讨论区】现在后端统一用 topic 字段存主题 ID
  if (notification.topic) {
    return `https://zhidianworld.com/topic/?id=${notification.topic}${anchor}`;
  }

  // 4️⃣ 兼容旧字段（如果以后你又加回 topicId 之类的）
  if (notification.topicId) {
    return `https://zhidianworld.com/topic/?id=${notification.topicId}${anchor}`;
  }

  // 5️⃣ 【作品区】尝试按常见命名推断
  if (notification.work) {
    return `https://zhidianworld.com/read/?id=${notification.work}${anchor}`;
  }
  if (notification.workId) {
    return `https://zhidianworld.com/read/?id=${notification.workId}${anchor}`;
  }

  // 6️⃣ 兜底：没有 topic，又有 likedWork 的情况，可能是作品
  // 注意：你的讨论区点赞也有 likedWork 字段，但那种情况 topic 也存在，
  // 所以上面已经 return 了，这里只在没有 topic 时当成作品处理。
  if (!notification.topic && notification.likedWork) {
    return `https://zhidianworld.com/read/?id=${notification.likedWork}${anchor}`;
  }

  // 7️⃣ 实在拼不出链接，返回 null，外层会提示“找不到”
  return null;
}
