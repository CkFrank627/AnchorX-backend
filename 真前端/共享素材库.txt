async function handleDownload(event) {
    event.stopPropagation();
    const galleryId = event.target.dataset.id;
    const galleryTitle = event.target.dataset.title || 'images';
    const button = event.target.closest('.download-btn');
    const originalContent = button.innerHTML;

    button.disabled = true;
    button.innerHTML = '<span class="loading-spinner"></span>';

    // 🟦 获取进度条 DOM
    const progressContainer = document.getElementById('download-progress-container');
    const progressText = document.getElementById('download-progress-text');
    const progressBar = document.getElementById('download-progress-bar');
    progressContainer.style.display = 'block';
    progressText.textContent = '正在准备下载...';
    progressBar.style.width = '0%';

    try {
        const response = await fetch(`${API_BASE_URL}/api/galleries/${galleryId}`);
        if (!response.ok) throw new Error('获取图库数据失败。');
        const gallery = await response.json();

        if (!gallery.images || gallery.images.length === 0) {
            alert('该图库没有图片可以下载。');
            progressContainer.style.display = 'none';
            return;
        }

        const zip = new JSZip();
        const totalImages = gallery.images.length;
        let downloadedCount = 0;

        for (const image of gallery.images) {
            const imageUrl = image.url;
            const imageResponse = await fetch(imageUrl);
            const blob = await imageResponse.blob();
            const filename = imageUrl.substring(imageUrl.lastIndexOf('/') + 1);
            zip.file(filename, blob, { binary: true });

            downloadedCount++;
            const percent = Math.round((downloadedCount / totalImages) * 100);
            progressText.textContent = `正在下载图片：${downloadedCount} / ${totalImages} (${percent}%)`;
            progressBar.style.width = `${percent}%`;
        }

        progressText.textContent = '正在打包 ZIP 文件...';
        progressBar.style.width = '100%';

        const content = await zip.generateAsync({ type: 'blob' });
        saveAs(content, `${galleryTitle}.zip`);

        progressText.textContent = '下载已完成！';
        progressBar.style.background = '#4caf50';
        setTimeout(() => {
            progressContainer.style.display = 'none';
            progressBar.style.background = '#0077cc';
        }, 1500);

    } catch (error) {
        console.error('下载图库失败:', error);
        alert('下载失败，请稍后重试。');
        progressContainer.style.display = 'none';
    } finally {
        button.disabled = false;
        button.innerHTML = originalContent;
    }
}

    const concurrency = 3;
    const queue = Array.from(files);
    const compressedFiles = [];
    let active = 0;

    async function processNext() {
        if (cancelRequested || queue.length === 0) return;
        const file = queue.shift();
        const index = compressedFiles.length;

        active++;
        try {
            const options = {
                maxSizeMB: 0.5,
                maxWidthOrHeight: 1920,
                useWebWorker: true,
                onProgress: (progress) => {
                    // 可选：单文件压缩时局部进度
                }
            };

            const compressedFile = await imageCompression(file, options);
            compressedFiles.push(compressedFile);
            processedCount++;

            const percent = Math.round((processedCount / totalFiles) * 100);
            globalBar.style.width = `${percent}%`;
            globalText.textContent = `压缩中：${processedCount} / ${totalFiles} (${percent}%)`;

            console.log(`✅ 压缩完成: ${file.name} (${(file.size/1024).toFixed(1)}KB → ${(compressedFile.size/1024).toFixed(1)}KB)`);

        } catch (err) {
            console.error(`❌ 压缩失败: ${file.name}`, err);
        } finally {
            active--;
            if (!cancelRequested && queue.length > 0) processNext();
        }
    }

// 同时启动 3 个任务
for (let i = 0; i < concurrency; i++) {
    if (i < queue.length) {
        processNext();
    } else {
        break;
    }
}

        new Promise(resolve => {
            const check = setInterval(() => {
                if (cancelRequested || (queue.length === 0 && active === 0)) {

                    resolve();
                }

    function renderPreviews() {
        imagePreviewsContainer.innerHTML = '';
        coverThumbnailsContainer.innerHTML = '';
        uploadedFiles.forEach((file, index) => {
            const reader = new FileReader();

                imagePreviewsContainer.appendChild(previewItem);

        });

    // 辅助函数：创建单个图片预览项
    function createPreviewItem(imgUrl, index) {
        const previewItem = document.createElement('div');
        previewItem.className = 'image-preview-item';
        if (index === selectedCoverIndex) {
            previewItem.classList.add('selected');
        }
        previewItem.innerHTML = `<img src="${imgUrl}" alt="预览">
                                 <button class="remove-btn" data-index="${index}">&times;</button>`;
        previewItem.addEventListener('click', () => {
            selectedCoverIndex = index;
            renderPreviews();
        });
        const removeBtn = previewItem.querySelector('.remove-btn');
        removeBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            uploadedFiles.splice(index, 1);
            if (index === selectedCoverIndex) {
                selectedCoverIndex = uploadedFiles.length > 0 ? 0 : -1;
            } else if (index < selectedCoverIndex) {
                selectedCoverIndex--;
            }
            renderPreviews();
        });
        return previewItem;
    }

async function uploadFileToImgBB(file) {
  try {
    // 压缩逻辑
    const compressedFile = await imageCompression(file, { maxSizeMB: 0.5, maxWidthOrHeight: 1920, useWebWorker: true });
    console.log(`上传压缩版: ${file.name} (${(file.size / 1024).toFixed(1)} KB → ${(compressedFile.size / 1024).toFixed(1)} KB)`);

    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onloadend = async () => {
        const base64Image = reader.result.split(',')[1];
        const formData = new FormData();
        formData.append('image', base64Image);
        try {
          const response = await fetch('https://api.imgbb.com/1/upload?key=040dbc9e6e680f321e09d9e05f447094', {
            method: 'POST',
            body: formData
          });
          const data = await response.json();
          if (data.success) {
            resolve(data.data.url);
          } else {
            reject(new Error('图床上传失败: ' + data.error.message));
          }
        } catch (err) {
          reject(err);
        }
      };
      reader.readAsDataURL(compressedFile);
    });
  } catch (err) {
    console.error('图片压缩失败或上传到ImgBB失败:', err);
    throw err;
  }
}
