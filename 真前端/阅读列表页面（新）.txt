  document.addEventListener('DOMContentLoaded', function () {

      page: 1,

      hasMore: true,
      loading: false,

    var toastTimer = null;

    function escapeHtml(value) {
      return String(value == null ? '' : value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function showToast(message) {
      toast.textContent = message;
      toast.classList.add('show');
      if (toastTimer) clearTimeout(toastTimer);
      toastTimer = setTimeout(function () { toast.classList.remove('show'); }, 2600);
    }

    function applyPageTheme(theme) {
      var allowedThemes = ['winter', 'plum', 'forest', 'sunset'];
      if (allowedThemes.indexOf(theme) === -1) theme = 'winter';
      document.body.setAttribute('data-theme', theme);
      localStorage.setItem('reading-list-theme', theme);
      var buttons = themeOptions.querySelectorAll('[data-theme]');
      for (var i = 0; i < buttons.length; i += 1) {
        var isActive = buttons[i].getAttribute('data-theme') === theme;
        buttons[i].classList.toggle('active', isActive);
        buttons[i].setAttribute('aria-pressed', isActive ? 'true' : 'false');
      }
    }

    function formatNumber(value) {
      var number = Number(value) || 0;
      if (number >= 100000000) return (number / 100000000).toFixed(1).replace('.0', '') + '亿';
      if (number >= 10000) return (number / 10000).toFixed(1).replace('.0', '') + '万';
      return number.toLocaleString('zh-CN');
    }

    function formatDate(value) {
      if (!value) return '日期未知';
      var date = new Date(value);
      if (Number.isNaN(date.getTime())) return '日期未知';
      return date.toLocaleDateString('zh-CN', { year: 'numeric', month: 'short', day: 'numeric' });
    }

    function getWorkId(work) {
      return work ? (work._id ? String(work._id) : '') : '';
    }

    function getAuthor(work) {
      if (work) {
        if (work.author) {
          if (work.author.username) return String(work.author.username);
        }
      }
      return '匿名作者';
    }

    function getTags(work) {
      if (!work) return [];
      if (!Array.isArray(work.tags)) return [];
      return work.tags.filter(function (tag) { return typeof tag === 'string' ? Boolean(tag.trim()) : false; });
    }

    function getSummary(work) {
      if (work) {
        var candidates = [work.summary, work.description, work.introduction];
        for (var i = 0; i < candidates.length; i += 1) {
          if (typeof candidates[i] === 'string') {
            var text = candidates[i].replace(/<[^>]*>/g, '').trim();
            if (text) return text.slice(0, 180);
          }
        }
      }
      return '作者似乎还没有为这部作品写下简介……';
    }

    function workHtml(work) {
      var id = getWorkId(work);
      var href = READ_URL + encodeURIComponent(id);
      var title = work ? (work.title ? work.title : '未命名作品') : '未命名作品';
      var cover = work ? (work.coverImage ? work.coverImage : PLACEHOLDER_COVER) : PLACEHOLDER_COVER;
      var updatedAt = work ? (work.updatedAt || work.createdAt) : null;
      var tags = getTags(work).slice(0, 4);
      var tagsHtml = '';
      for (var i = 0; i < tags.length; i += 1) {
        tagsHtml += '<span class="work-tag">' + escapeHtml(tags[i]) + '</span>';
      }
      if (!tagsHtml) tagsHtml = '<span class="work-tag">未分类</span>';

      return ''
        + '<article class="work-item">'
        + '  <a class="work-cover-link" href="' + href + '" aria-label="阅读《' + escapeHtml(title) + '》">'
        + '    <img class="work-cover" src="' + escapeHtml(cover) + '" alt="《' + escapeHtml(title) + '》封面" loading="lazy" />'
        + '  </a>'
        + '  <div class="work-body">'
        + '    <h3 class="work-title"><a href="' + href + '">' + escapeHtml(title) + '</a></h3>'
        + '    <div class="work-meta"><span>✎ ' + escapeHtml(getAuthor(work)) + '</span><span>更新于 ' + escapeHtml(formatDate(updatedAt)) + '</span><span>' + formatNumber(work ? work.wordCount : 0) + ' 字</span></div>'
        + '    <p class="work-summary">' + escapeHtml(getSummary(work)) + '</p>'
        + '    <div class="work-footer">'
        + '      <div class="work-tags">' + tagsHtml + '</div>'
        + '      <div class="work-stats"><span>◉ ' + formatNumber(work ? work.views : 0) + '</span><span>♥ ' + formatNumber(work ? work.likesCount : 0) + '</span></div>'
        + '    </div>'
        + '  </div>'
        + '</article>';
    }

    function renderWorks() {
      worksList.classList.toggle('grid-mode', state.view === 'grid');
      listViewBtn.classList.toggle('active', state.view === 'list');
      gridViewBtn.classList.toggle('active', state.view === 'grid');
      listViewBtn.setAttribute('aria-pressed', state.view === 'list' ? 'true' : 'false');
      gridViewBtn.setAttribute('aria-pressed', state.view === 'grid' ? 'true' : 'false');

      if (!state.works.length) {
        if (!state.loading) worksList.innerHTML = '<div class="list-state">哎呀，没有找到符合条件的作品，试试别的吧。</div>';
        return;
      }
      var html = [];
      for (var i = 0; i < state.works.length; i += 1) html.push(workHtml(state.works[i]));
      worksList.innerHTML = html.join('');
      renderFeatures();
    }

    function renderActiveFilter() {
      var parts = [];
      if (state.query) parts.push('名称含“' + state.query + '”');
      if (state.tag) parts.push('分类：' + state.tag);
      if (state.sort !== 'updated') {
        var labels = { newest: '最新发布', views: '阅读最多', likes: '喜欢最多' };
        parts.push('排序：' + labels[state.sort]);
      }
      activeFilter.hidden = parts.length === 0;
      activeFilter.textContent = parts.length ? '当前筛选 · ' + parts.join(' · ') : '';
      filterToggle.classList.toggle('active', parts.length > 0);
    }

    function setFilterOpen(open) {
      filterPopover.hidden = !open;
      filterToggle.setAttribute('aria-expanded', open ? 'true' : 'false');
      if (open) setTimeout(function () { searchInput.focus(); }, 0);
    }

    function buildWorksUrl() {
      var params = new URLSearchParams();
      params.set('page', String(state.page));
      params.set('limit', String(state.limit));
      params.set('sort', state.sort);
      if (state.query) params.set('q', state.query);
      if (state.tag) params.set('tag', state.tag);
      return WORKS_API + '?' + params.toString();
    }

    function updateLoadMore() {
      loadMoreBtn.disabled = state.loading || !state.hasMore;
      if (state.loading) loadMoreBtn.textContent = '正在加载……';
      else if (state.hasMore) loadMoreBtn.textContent = '继续加载';
      else loadMoreBtn.textContent = state.works.length ? '已经到底了' : '暂无更多作品';
    }

    function loadWorks(reset) {
      if (state.loading) return;
      if (!reset) {
        if (!state.hasMore) return;
      }
      if (reset) {
        state.page = 1;
        state.hasMore = true;
        state.works = [];
        worksList.innerHTML = '<div class="list-state">正在按新的条件整理作品……</div>';
      }
      state.loading = true;
      updateLoadMore();

      fetch(buildWorksUrl())
        .then(function (response) {
          if (!response.ok) throw new Error('作品列表请求失败');
          return response.json();
        })
        .then(function (data) {
          var incoming = data ? (Array.isArray(data.works) ? data.works : []) : [];
          state.works = state.works.concat(incoming);
          state.total = data ? (typeof data.total === 'number' ? data.total : state.works.length) : state.works.length;
          state.hasMore = data ? Boolean(data.hasMore) : false;
          state.page += 1;
          workCount.textContent = formatNumber(state.total);
        })
        .catch(function (error) {
          if (!state.works.length) worksList.innerHTML = '<div class="list-state">作品暂时无法正常加载，请稍后再试，如果重复出问题，请拷打站长罢！</div>';
          showToast(error.message || '作品加载失败');
        })
        .then(function () {
          state.loading = false;
          renderWorks();
          updateLoadMore();
          renderActiveFilter();
        });
    }

    function miniWorksHtml(works) {
      if (!works.length) return '<p class="empty-mini">暂时没有匹配的作品。</p>';
      var html = [];
      for (var i = 0; i < works.length; i += 1) {
        var work = works[i];
        html.push('<a class="mini-work" href="' + READ_URL + encodeURIComponent(getWorkId(work)) + '"><strong>' + escapeHtml(work.title || '未命名作品') + '</strong><span>' + escapeHtml(getAuthor(work)) + ' · ' + formatNumber(work.views) + ' 阅读</span></a>');
      }
      return html.join('');
    }

    function renderFeatures() {
      if (!state.works.length) return;
      var winter = state.works.filter(function (work) {
        return getTags(work).some(function (tag) { return /冬|雪|新年|节日/.test(tag); });
      }).slice(0, 3);
      if (!winter.length) winter = state.works.slice(0, 3);

      var monthly = state.works.slice().sort(function (a, b) {
        return (Number(b.likesCount) || 0) - (Number(a.likesCount) || 0);
      }).slice(0, 3);
      document.getElementById('winterList').innerHTML = miniWorksHtml(winter);
      document.getElementById('monthlyList').innerHTML = miniWorksHtml(monthly);
    }

    function applyTag(tag) {
      state.tag = tag || '';
      tagSelect.value = state.tag;
      renderCategories();
      setFilterOpen(false);
      loadWorks(true);
      document.getElementById('worksHeading').scrollIntoView({ behavior: 'smooth', block: 'start' });
    }

    function renderCategories() {
      if (!state.tags.length) {
        categoryList.innerHTML = '<span class="empty-mini">暂无可用分类。</span>';
        return;
      }
      var html = ['<button class="category-chip' + (state.tag ? '' : ' active') + '" type="button" data-tag="">全部</button>'];
      for (var i = 0; i < state.tags.length; i += 1) {
        var item = state.tags[i];
        var active = state.tag === item.tag ? ' active' : '';
        html.push('<button class="category-chip' + active + '" type="button" data-tag="' + escapeHtml(item.tag) + '">' + escapeHtml(item.tag) + ' <span>' + formatNumber(item.count) + '</span></button>');
      }
      categoryList.innerHTML = html.join('');
    }

    function loadTags() {
      fetch(TAGS_API)
        .then(function (response) {
          if (!response.ok) throw new Error('分类加载失败');
          return response.json();
        })
        .then(function (data) {
          state.tags = data ? (Array.isArray(data.tags) ? data.tags.slice(0, 18) : []) : [];
          var options = ['<option value="">全部分类</option>'];
          for (var i = 0; i < state.tags.length; i += 1) {
            options.push('<option value="' + escapeHtml(state.tags[i].tag) + '">' + escapeHtml(state.tags[i].tag) + '（' + formatNumber(state.tags[i].count) + '）</option>');
          }
          tagSelect.innerHTML = options.join('');
          tagSelect.value = state.tag;
          renderCategories();
        })
        .catch(function () {
          categoryList.innerHTML = '<span class="empty-mini">分类暂时无法加载。</span>';
        });
    }

    function loadStats() {
      fetch(STATS_API)
        .then(function (response) {
          if (!response.ok) throw new Error('统计加载失败');
          return response.json();

        .then(function (data) {
          document.getElementById('statWorks').textContent = formatNumber(data.works);
          document.getElementById('statAuthors').textContent = formatNumber(data.authors);
          document.getElementById('statViews').textContent = formatNumber(data.views);

        .catch(function () {
          var boxes = document.querySelectorAll('.stat-box strong');

    function loadTopics() {
      fetch(TOPICS_API)
        .then(function (response) {
          if (!response.ok) throw new Error('讨论加载失败');
          return response.json();
        })
        .then(function (data) {
          var topics = data ? (Array.isArray(data.topics) ? data.topics : []) : [];
          if (!topics.length) {
            topicList.innerHTML = '<p class="empty-mini">还没有新的讨论。</p>';
            return;
          }
          var html = [];
          for (var i = 0; i < topics.length; i += 1) {
            var topic = topics[i];
            html.push('<a class="topic-item" href="/topic/?id=' + encodeURIComponent(topic._id) + '"><span class="topic-title">' + escapeHtml(topic.title || '未命名主题') + '</span><span class="topic-meta">' + formatNumber(topic.repliesCount) + ' 条回复</span></a>');
          }
          topicList.innerHTML = html.join('');
        })
        .catch(function () {
          topicList.innerHTML = '<p class="empty-mini">讨论区暂时无法连接。</p>';
        });
    }

    function renderCheckIn(data) {
      var checked = data ? Boolean(data.checkedInToday) : false;
      checkInBtn.classList.toggle('is-done', checked);
      checkInBtn.textContent = checked ? '✓ 今日已签到' : '📅 今日签到';
      checkInBtn.disabled = checked;
      var streak = data ? Number(data.streak) || 0 : 0;
      var total = data ? Number(data.totalDays) || 0 : 0;
      checkInDetail.textContent = '连续 ' + streak + ' 天 · 累计 ' + total + ' 天';
    }

    function loadCheckIn() {
      var token = localStorage.getItem('token');
      if (!token) {
        checkInBtn.textContent = '登录后签到';
        checkInDetail.textContent = '每日签到会记录连续天数';
        return;
      }
      fetch(CHECK_IN_API, { headers: { Authorization: 'Bearer ' + token } })
        .then(function (response) {
          if (response.status === 401) throw new Error('登录已过期');
          if (!response.ok) throw new Error('签到状态加载失败，请拷打站长罢');
          return response.json();
        })
        .then(renderCheckIn)
        .catch(function (error) {
          checkInDetail.textContent = error.message;
        });
    }

    checkInBtn.addEventListener('click', function () {
      var token = localStorage.getItem('token');
      if (!token) {
        window.location.href = LOGIN_URL;
        return;
      }
      checkInBtn.disabled = true;
      checkInBtn.textContent = '正在签到……';
      fetch(CHECK_IN_API, {
        method: 'POST',
        headers: { Authorization: 'Bearer ' + token, 'Content-Type': 'application/json' },
        body: '{}'
      })
        .then(function (response) {
          return response.json().then(function (data) {
            if (!response.ok) throw new Error(data.message || '签到失败');
            return data;
          });
        })
        .then(function (data) {
          renderCheckIn(data);
          showToast(data.message || '签到成功');
        })
        .catch(function (error) {
          checkInBtn.disabled = false;
          checkInBtn.textContent = '📅 重新签到';
          showToast(error.message || '签到失败');
        });
    });

    function createSnow() {
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
      var field = document.getElementById('snowfield');
      for (var i = 0; i < 24; i += 1) {
