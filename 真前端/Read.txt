// 初始化 IntersectionObserver（在句子到达视口上方 1/5 附近触发）
// ---------------------------------------------------------------------
// 重写 initEffectObserver —— 触发条件改成“句子到达屏幕下方 1/5”
// ---------------------------------------------------------------------
function initEffectObserver() {
    if (!('IntersectionObserver' in window)) {
        if (EFFECT_DEBUG) {
            console.log('⚠️ [effects] 当前环境不支持 IntersectionObserver，改用滚动监听方案');
        }
        return null;
    }

    // 计算视口高度，用来确定“下方 1/5”这条带状区域
    var viewportHeight = window.innerHeight || document.documentElement.clientHeight || 0;
    // 下方 1/5 的上边界，例如 0.8 * height
    var lowerBandTop = viewportHeight * 0.8;
    // 下方 1/5 的下边界就是视口底部
    var lowerBandBottom = viewportHeight;

    if (EFFECT_DEBUG) {
        console.log(
            '🔧 [effects] IntersectionObserver 初始化完成：\n',
            '   viewportHeight =', viewportHeight,
            '   lowerBandTop =', lowerBandTop,
            '   lowerBandBottom =', lowerBandBottom
        );
    }

    // 构造一组 threshold，让元素在视口内部移动时也能多次触发回调
    var thresholds = [];
    var t = 0.0;
    while (t <= 1.0) {
        thresholds.push(t);
        t = t + 0.1;     // 每 0.1 一个点，你也可以改成 0.05 更细
    }

    effectObserver = new IntersectionObserver(function (entries) {
        entries.forEach(function (entry) {
            var target = entry.target;
            var cb = target.__effectTrigger;
            if (!cb) return;

            var rect = target.getBoundingClientRect();
            var shouldTrigger = false;

            // 用元素中心点判断
            var centerY = rect.top + rect.height / 2;
            if (centerY >= lowerBandTop) {
                if (centerY <= lowerBandBottom) {
                    shouldTrigger = true;
                }
            }

            if (!shouldTrigger) {
                return;
            }

            if (EFFECT_DEBUG) {
                console.log(
                    '🚀 [IO] 触发特效：',
                    target.getAttribute('data-sentence-id'),
                    ' rect.top =', rect.top.toFixed(1),
                    ' rect.bottom =', rect.bottom.toFixed(1),
                    ' centerY =', centerY.toFixed(1),
                    ' lowerBandTop =', lowerBandTop.toFixed(1),
                    ' lowerBandBottom =', lowerBandBottom.toFixed(1),
                    ' ratio =', entry.intersectionRatio.toFixed(2)
                );
            }

            try {
                cb(target);
            } catch (err) {
                console.error('执行特效时出错:', err);
            }

            // 只触发一次就取消监听
            effectObserver.unobserve(target);
            delete target.__effectTrigger;
        });
    }, {
        root: null,
        threshold: thresholds
    });


    return effectObserver;
}

function registerEffectTrigger(targetEl, cb) {
    if (!targetEl || typeof cb !== 'function') return;

    const sid = targetEl.getAttribute('data-sentence-id') || '(no-id)';

    if (EFFECT_DEBUG) {
    var useObserver = false;

    // 先判断是否启用 Observer
    if (EFFECT_USE_OBSERVER) {
        // 再判断浏览器是否支持 IntersectionObserver
        if ('IntersectionObserver' in window) {
            useObserver = true;
        }
    }

    console.log(
        '🧪 [registerEffectTrigger] 目标句子 =',
        sid,
        '，当前在 DOM 中 =',
        document.body.contains(targetEl),
        '，使用 Observer =',
        useObserver
    );
}


    // 调试阶段：如果关闭懒加载，或者浏览器不支持 IO，就直接执行一次特效
    if (!EFFECT_USE_OBSERVER || !('IntersectionObserver' in window)) {
        try {
            cb(targetEl);
        } catch (err) {
            console.error('❌ [registerEffectTrigger] 直接执行特效出错:', sid, err);
        }
        return;
    }

            initEffectObserver();
    if (!effectObserver) {
        try {
            cb(targetEl);
        } catch (err) {
            console.error('❌ [registerEffectTrigger] effectObserver 不存在，fallback 执行特效失败:', sid, err);
        }
        return;
    }

    // ⭐ 新增：顶部补触发逻辑（兼容旧转换器版本）
    (function () {
        var rect = targetEl.getBoundingClientRect();
        var viewportHeight = window.innerHeight || document.documentElement.clientHeight || 0;
        var lowerBandTop = viewportHeight * 0.8;   // 保持 0.8 不变
        var lowerBandBottom = viewportHeight;

        // 计算是否在视口内（使用嵌套判断）
        var inViewport = false;
        if (rect.bottom > 0) {
            if (rect.top < viewportHeight) {
                inViewport = true;
            }
        }

        var centerY = rect.top + rect.height / 2;

        // 如果当前已经在视口里，但中心点在下方 1/5 以上，直接播一次特效
        if (inViewport) {
            if (centerY < lowerBandTop) {
                if (EFFECT_DEBUG) {
                    console.log(
                        '⚡ [registerEffectTrigger] 顶部补触发：',
                        sid,
                        ' rect.top =', rect.top.toFixed(1),
                        ' rect.bottom =', rect.bottom.toFixed(1),
                        ' centerY =', centerY.toFixed(1),
                        ' lowerBandTop =', lowerBandTop.toFixed(1),
                        ' lowerBandBottom =', lowerBandBottom.toFixed(1)
                    );
                }
                try {
                    cb(targetEl);
                } catch (err) {
                    console.error('❌ [registerEffectTrigger] 顶部补触发失败:', sid, err);
                }
                // 不再交给 Observer
                return;
            }
        }
    })();

    // 正常情况：交给 IntersectionObserver，等它滚到屏幕下方 1/5 再触发
    targetEl.__effectTrigger = cb;
    effectObserver.observe(targetEl);
}

      function sanitizeEffectCode(raw) {
  if (raw == null) return '';
  var s = String(raw);

  // 统一换行
  s = s.replace(/\r\n/g, '\n').replace(/\r/g, '\n');

  // 去 BOM / 零宽字符 / 软连字符
  if (s.charCodeAt(0) === 0xFEFF) s = s.slice(1);
  s = s.replace(/[\u200B-\u200D\u2060\u00AD]/g, '');
  s = s.replace(/\u00A0/g, ' ');

  // 某些环境会把 U+2028/U+2029 当“非法换行”
  s = s.replace(/\u2028/g, '\n').replace(/\u2029/g, '\n');

  // 智能引号 → 普通引号（特别常见：从富文本/文档里复制代码）
  s = s.replace(/[“”]/g, '"').replace(/[‘’]/g, "'");

  // 常见全角标点（可选但建议）
  s = s.replace(/；/g, ';').replace(/，/g, ',').replace(/（/g, '(').replace(/）/g, ')');

  // 如果明显像被 HTML/富文本包过，尝试“还原成纯文本”
  var looksHtml = false;
  if (s.indexOf('&lt;') !== -1) looksHtml = true;
  if (s.indexOf('&gt;') !== -1) looksHtml = true;
  if (s.indexOf('&nbsp;') !== -1) looksHtml = true;
  if (s.indexOf('<br') !== -1) looksHtml = true;
  if (s.indexOf('<div') !== -1) looksHtml = true;
  if (looksHtml) {
    var div = document.createElement('div');
    div.innerHTML = s;
    // innerText 比 textContent 更像“复制出来的代码”，会保留换行
    s = div.innerText || div.textContent || s;
  }

  // 兼容从 Markdown 或完整 HTML 示例中复制的代码。
  var fenced = s.trim().match(/^```(?:javascript|js)?\s*([\s\S]*?)\s*```$/i);
  if (fenced) s = fenced[1];
  var scriptWrapped = s.trim().match(/^<script(?:\s[^>]*)?>\s*([\s\S]*?)\s*<\/script>$/i);
  if (scriptWrapped) s = scriptWrapped[1];

  return s;
}

function findFirstSuspiciousChar(raw) {
  if (raw == null) return null;
  var s = String(raw);
  for (var i = 0; i < s.length; i++) {
    var code = s.charCodeAt(i);

    // 控制字符：除 tab / \n / \r 外都可疑
    if (code < 32) {
      if (code !== 9) {
  if (code !== 10) {
    if (code !== 13) {
      return { index: i, code: code };
    }
  }
}

    }

    // BOM / 行分隔符 / 零宽
    if (code === 0xFEFF || code === 0x2028 || code === 0x2029) return { index: i, code: code };
    if (code >= 0x200B ? code <= 0x200D : false) return { index: i, code: code };
    if (code === 0x2060 || code === 0x00AD) return { index: i, code: code };
  }
  return null;
}

        code + '\n' +
        'if (typeof applyEffect === "function") { return applyEffect(targetEl); }\n' +

        srcURL;

      console.warn('⚠️ [effects] 特效在非 strict 模式下才编译通过:', key);

      });
    }
  }

async function ensurePagesFetchedUpTo(targetEnd, weak = false) {
  const workId = getActiveWorkIdForRead();
  if (!workId) return;

  while (pagesFetchCursor < targetEnd) {
    if (pagesFetchInFlight) {
      await pagesFetchInFlight;
      continue;
    }

    const start = pagesFetchCursor;
    const batch = weak ? WEAK_INITIAL_PAGES : FLOOR_BATCH_SIZE;
    const limit = Math.min(batch, targetEnd - start);

    pagesFetchInFlight = (async () => {
      const data = await fetchWorkPagesBatch(workId, start, limit, weak);
      const pages = Array.isArray(data.pages) ? data.pages : [];

      // ✅ 兼容后端返回 pageCount（以及你未来可能改成 totalPages）
      const hasTotalPages =
        typeof data.totalPages === "number" ? data.totalPages > 0 : false;
      const hasPageCount =
        typeof data.pageCount === "number" ? data.pageCount > 0 : false;
      const serverTotal = hasTotalPages
        ? data.totalPages
        : (hasPageCount ? data.pageCount : 0);

      if (serverTotal) {
        totalPageCount = serverTotal;
        if (!Array.isArray(activePages) || activePages.length !== totalPageCount) {
          const old = Array.isArray(activePages) ? activePages : [];
          activePages = new Array(totalPageCount).fill(null);
          for (let i = 0; i < Math.min(old.length, activePages.length); i++) {
            if (old[i]) activePages[i] = old[i];
          }
        }
      }

      for (let i = 0; i < pages.length; i++) {
        const p = pages[i] || {};
        const idx = (typeof p.index === "number") ? p.index : (start + i);

        if (!activePages[idx]) activePages[idx] = {};
        activePages[idx].content = p.content || { ops: [] };

        if (typeof p.sentenceStartIndex === "number") {
          activePages[idx].sentenceStartIndex = p.sentenceStartIndex;
        }
      }

      pagesFetchCursor = start + pages.length;
    })().finally(() => {
      pagesFetchInFlight = null;
    });

    await pagesFetchInFlight;
  }
}

function computePageHasAnyComment(pageIndex, workId) {
  if (!commentedIdsReady) return null;

  const page = activePages[pageIndex];
  if (!page) return null;

  const start = page.sentenceStartIndex;
  const count = page.__sentenceIdsCount;

  if (typeof start !== 'number') return null;
  if (typeof count !== 'number') return null;

  for (let k = 0; k < count; k++) {
    const sid = buildSentenceId(workId, start + k);
    if (commentedSentenceIds.has(sid)) return true;
  }
  return false;
}

function recomputeCommentCountForPage(pageIndex, workId) {
  const page = activePages[pageIndex];
  if (!page) return;

  const start = page.sentenceStartIndex;
  const count = page.__sentenceIdsCount;

  if (typeof start !== 'number') return;
  if (typeof count !== 'number') return;

  let total = 0;
  let needEllipsis = false;
  let anyCommented = false;

  for (let k = 0; k < count; k++) {
    const sid = buildSentenceId(workId, start + k);
    if (commentedSentenceIds.has(sid)) {
      anyCommented = true;

      if (loadedCommentSentenceIds.has(sid)) {
        const arr = commentBySentenceId.get(sid);
        if (Array.isArray(arr)) total += arr.length;
      } else {
        needEllipsis = true;
      }
    }
  }

  if (!anyCommented) {
    commentCountByPage.set(pageIndex, 0);
    return;
  }

  if (needEllipsis) {
    commentCountByPage.delete(pageIndex);
    return;
  }

  commentCountByPage.set(pageIndex, total);
}

function chunkArray(arr, size) {
  const out = [];
  let i = 0;
  while (i < arr.length) {
    out.push(arr.slice(i, i + size));
    i += size;
  }
  return out;
}

async function ensureCommentsLoadedForPagesRange(startPage, endPage) {
  const workId = getActiveWorkIdForComments();
  if (!workId) return;

  if (commentLoadedWorkId !== workId) {
    resetCommentCacheForWork(workId);
  }

  await ensureCommentedIdsReadyNow();

  if (!Array.isArray(activePages)) return;

  const idsToFetch = [];
  const pagesTouched = new Set();

  for (let p = startPage; p < endPage; p++) {
    pagesTouched.add(p);

    const page = activePages[p];
    if (!page) continue;

    const start = page.sentenceStartIndex;
    const count = page.__sentenceIdsCount;

    if (typeof start !== 'number') continue;
    if (typeof count !== 'number') continue;

    for (let k = 0; k < count; k++) {
      const sid = buildSentenceId(workId, start + k);

      if (commentedSentenceIds.has(sid)) {
        if (!loadedCommentSentenceIds.has(sid)) {
          idsToFetch.push(sid);
        }
      }
    }
  }

  const chunks = chunkArray(idsToFetch, 200);

  for (let c = 0; c < chunks.length; c++) {
    const part = chunks[c];
    if (part.length === 0) continue;

    const rows = await fetchCommentsBySentenceIds(workId, part);

    const grouped = new Map();
    for (let i = 0; i < rows.length; i++) {
      const item = rows[i];
      const sid = item ? item.sentenceId : null;
      if (!sid) continue;

      if (!grouped.has(sid)) grouped.set(sid, []);
      grouped.get(sid).push(item);
    }

    grouped.forEach((arr, sid) => {
      commentBySentenceId.set(sid, arr);
      loadedCommentSentenceIds.add(sid);
    });

    for (let i = 0; i < part.length; i++) {
      const sid = part[i];
      loadedCommentSentenceIds.add(sid);
      if (!commentBySentenceId.has(sid)) commentBySentenceId.set(sid, []);
    }
  }

  pagesTouched.forEach(p => {
    recomputeCommentCountForPage(p, workId);
  });

  updateFloorCommentButtonsRange(startPage, endPage);
}

async function ensureCommentsLoadedForSentenceId(sentenceId) {
  const workId = getActiveWorkIdForComments();
  if (!workId) return;

  if (commentLoadedWorkId !== workId) {

  }

  await ensureCommentedIdsReadyNow();

  if (loadedCommentSentenceIds.has(sentenceId)) return;

  const rows = await fetchCommentsBySentenceIds(workId, [sentenceId]);

  const arr = [];
  for (let i = 0; i < rows.length; i++) {
    const it = rows[i];
    if (!it) continue;
    if (it.sentenceId === sentenceId) arr.push(it);
  }

  commentBySentenceId.set(sentenceId, arr);
  loadedCommentSentenceIds.add(sentenceId);

  const el = document.querySelector(`[data-sentence-id="${sentenceId}"]`);
  if (el) {
    const floor = el.closest('.floor');
    if (floor) {
      const s = floor.getAttribute('data-page-index');
      const p = parseInt(s || '-1', 10);
      if (!Number.isNaN(p)) {
        recomputeCommentCountForPage(p, workId);
        updateFloorCommentButton(p);
      }
    }
  }
}

function normalizeSentenceAnchorText(value) {
    return String(value || '')
        .replace(/[\u200B-\u200D\uFEFF]/g, '')
        .replace(/\s+/g, ' ')
        .trim()
        .slice(0, 1000);
}

function resolveAnchoredSentenceElement(rootEl, rawIndex, anchor) {
    if (!rootEl) return null;

    const expectedIndex = Number(rawIndex);
    if (!Number.isInteger(expectedIndex) || expectedIndex < 0) return null;

    const expected = rootEl.querySelector(`[data-sentence-index="${expectedIndex}"]`);
    const anchorText = normalizeSentenceAnchorText(anchor ? anchor.text : '');
    if (!anchorText) return expected;

    const textOf = (el) => {
        if (!el) return normalizeSentenceAnchorText('');
        if (el.dataset) {
            if (el.dataset.sentenceAnchorText) {
                return normalizeSentenceAnchorText(el.dataset.sentenceAnchorText);
            }
        }
        return normalizeSentenceAnchorText(el.textContent);
    };
    if (expected ? textOf(expected) === anchorText : false) return expected;

    const beforeText = normalizeSentenceAnchorText(anchor ? anchor.before : '');
    const afterText = normalizeSentenceAnchorText(anchor ? anchor.after : '');
    const candidates = Array.from(rootEl.querySelectorAll('[data-sentence-index]'))
        .filter((el) => textOf(el) === anchorText);

    let best = null;
    let bestScore = -Infinity;
    candidates.forEach((candidate) => {
        const candidateIndex = Number(candidate.dataset.sentenceIndex);
        if (!Number.isInteger(candidateIndex)) return;

        const previous = rootEl.querySelector(`[data-sentence-index="${candidateIndex - 1}"]`);
        const next = rootEl.querySelector(`[data-sentence-index="${candidateIndex + 1}"]`);
        const beforeMatches = beforeText ? textOf(previous) === beforeText : false;
        const afterMatches = afterText ? textOf(next) === afterText : false;
        const hasContext = Boolean(beforeText || afterText);
        if (hasContext) {
            if (!beforeMatches) {
                if (!afterMatches) return;
            }
        }

        let score = 10 - Math.min(Math.abs(candidateIndex - expectedIndex), 100) / 100;
        if (beforeMatches) score += 4;
        if (afterMatches) score += 4;
        if (score > bestScore) {
            best = candidate;
            bestScore = score;
        }
    });

    return best;
}

// ⭐ 工具函数：为内容按“有效句子”编号并写入 data-sentence-id / data-sentence-index
function segmentContentAndAddIds(rawHtmlContent, workId, startIndex = 0) {
  const tempDiv = document.createElement('div');
  tempDiv.innerHTML = rawHtmlContent;
  let counter = startIndex;

  // 和 applyEffectsToBlocks 用同一套“行”的定义
  const blocks = tempDiv.querySelectorAll(
    'p, h1, h2, h3, li, blockquote, pre, .ql-code-block'
  );

  function assignSentenceData(el, text) {
    const normalizedText = normalizeSentenceAnchorText(text);
    if (!normalizedText) return;

    if (!el.hasAttribute('data-sentence-id')) {
      const sentenceId = 's_' + workId + '_' + counter;
      el.setAttribute('data-sentence-id', sentenceId);
    }
    el.dataset.sentenceId = el.getAttribute('data-sentence-id');
    el.dataset.sentenceIndex = String(counter);
    el.dataset.sentenceAnchorText = normalizedText;
    counter = counter + 1;
  }

  blocks.forEach(function (el) {
    if (!tempDiv.contains(el)) return;
    if (el.matches('.ql-code-block')) {
      assignSentenceData(el, el.textContent);
      return;
    }

    const rawText = el.textContent || '';
    if (el.tagName === 'PRE' ? /\r?\n/.test(rawText) : false) {
      const lines = rawText.split(/\r?\n/);
      el.textContent = '';
      lines.forEach(function (line, index) {
        if (normalizeSentenceAnchorText(line)) {
          const lineEl = document.createElement('span');
          lineEl.className = 'reader-code-line';
          lineEl.textContent = line;
          assignSentenceData(lineEl, line);
          el.appendChild(lineEl);
        }
        if (index + 1 < lines.length) el.appendChild(document.createTextNode('\n'));
      });
      return;
    }

    const text = rawText.trim();

    // 跳过纯空白
    if (text === '') {
      return;
    }
    assignSentenceData(el, text);
  });

  // ⭐ 调试：打印前 30 条句子的 行号 + ID + 文本片段
  if (window.READER_DEBUG_VERBOSE) {
    const sample = Array.from(tempDiv.querySelectorAll('[data-sentence-id]'))
      .slice(0, 30)
      .map(function (el) {
        return {
          index: el.dataset.sentenceIndex,
          id: el.dataset.sentenceId,
          text: el.textContent.trim().slice(0, 20)
        };
      });

    console.log('🧩 segmentContentAndAddIds 示例（前 30 条）:', sample);
  }

  return { element: tempDiv, nextIndex: counter };
}

async function ensurePagesPreparedUpTo(targetCount, { userInitiated = false } = {}) {
  if (!activeWork || !activeWork._id) return;
  if (!Array.isArray(activePages) || activePages.length === 0) return;

  const hardTarget = Math.min(targetCount, activePages.length);
  const myToken = pagePrepareToken;

  // 等待别的 prepare 结束
  while (pagePrepareRunning) {
    if (myToken !== pagePrepareToken) break;
    await sleep(40);
  }
  if (myToken !== pagePrepareToken) return;
  if (pagePrepareCursor >= hardTarget) return;

  pagePrepareRunning = true;
  const loopStart = Date.now();

  try {
    while (pagePrepareCursor < hardTarget) {
      if (myToken !== pagePrepareToken) return;

      // 弱网模式：如果不是用户触发的加载，只准备前两页就停
      if (isWeakLazyMode) {
  if (!userInitiated) {
    if (pagePrepareCursor >= WEAK_INITIAL_PAGES) {
      return;
    }
  }
}


      // ✅ 真懒加载关键：如果这一页还没拿到 content（可能是 null 占位 / 空对象），先去后端拉
      let page = activePages[pagePrepareCursor];
      const needFetch = !page || !page.content || !Array.isArray(page.content.ops);

      if (needFetch) {
        // 若你还没实现 ensurePagesFetchedUpTo，这里会直接 return（避免死循环）
        if (typeof ensurePagesFetchedUpTo !== 'function') {
          return;
        }

        // 只确保当前页（cursor 这页）拿到即可
        await ensurePagesFetchedUpTo(pagePrepareCursor + 1, { userInitiated });

        if (myToken !== pagePrepareToken) return;
        page = activePages[pagePrepareCursor];

        // 拉完依然没有，说明到头了/失败了，避免死循环
        if (!page || !page.content || !Array.isArray(page.content.ops)) {
          return;
        }
      }

      // ✅ 已准备过就跳过
      if (typeof page.sentenceStartIndex === 'number') {
        if (typeof page.segmentedHtml === 'string') {
          pagePrepareCursor++;
          continue;
        }
      }

      // ✅ 生成 sentenceStartIndex + segmentedHtml
      const startIndex = pageSentenceCounter;
      page.sentenceStartIndex = startIndex;

      const rawHtml = deltaToHtml(page.content || { ops: [] });
      const segmented = segmentContentAndAddIds(rawHtml, activeWork._id, startIndex);

// 记录本页“有效句子数量”（给评论计数与批量加载用）
let cnt = 0;
if (segmented) {
  if (typeof segmented.nextIndex === 'number') {
    cnt = segmented.nextIndex - page.sentenceStartIndex;
  }
}
if (typeof cnt !== 'number') cnt = 0;
if (cnt < 0) cnt = 0;
page.__sentenceIdsCount = cnt;

pageSentenceCounter = segmented.nextIndex;
page.segmentedHtml = segmented.element.innerHTML;

      pagePrepareCursor++;

      // 让出主线程，避免长循环卡死
      if (Date.now() - loopStart > 12) {
        await sleep(0);
      }
    }
  } finally {
    pagePrepareRunning = false;
  }
}

function startLazySlowWatchdog() {
  clearTimeout(lazySlowTimer);
  lazySlowTimer = setTimeout(() => {
    if (currentMode !== 'scroll') return;
    if (isWeakLazyMode) return;

    const need = Math.min(renderedFloorCount, activePages.length);

    var notReady = false;

    if (typeof pagePrepareCursor === 'number') {
      if (pagePrepareCursor < need) notReady = true;
    }

    // 适配“真·懒加载按需拉 pages”
    if (!notReady) {
      if (typeof pagesFetchCursor === 'number') {
        if (pagesFetchCursor < need) notReady = true;
      }
    }

    if (!notReady) {
      if (typeof pagesFetchInFlight !== 'undefined') {
        if (pagesFetchInFlight) notReady = true;
      }
    }

    if (notReady) {
      switchToWeakLazyMode();
    }
  }, LAZY_SLOW_THRESHOLD_MS);
}

function renderFloorsRange(startIndex, endIndex) {
  if (!activePages || activePages.length === 0) return;
  if (!bookContent) return;

  var didRenderHydrated = false;

  for (let pageIndex = startIndex; pageIndex < endIndex; pageIndex++) {
    if (pageIndex >= activePages.length) break;

    const page = activePages[pageIndex];

    const floorDiv = document.createElement('div');
    floorDiv.className = 'floor';
    floorDiv.dataset.pageIndex = String(pageIndex);

    const countText = getCommentsCountForPage(pageIndex);

    floorDiv.innerHTML = `
      <div class="floor-header">
        <span>${pageIndex + 1} 楼</span>
        <button class="comment-view-btn" data-page-index="${pageIndex}">
          查看本楼评论（${countText}）
        </button>
      </div>
      <div class="floor-body ql-editor" data-hydrated="0">
        ${page ? (page.segmentedHtml ? page.segmentedHtml : floorPlaceholderHtml()) : floorPlaceholderHtml()}
      </div>
    `;

    bookContent.appendChild(floorDiv);

    const floorBody = floorDiv.querySelector('.floor-body');

    if (commentBtn) {
      commentBtn.addEventListener('click', () => {

      });

    if (page) {
      if (page.segmentedHtml) {
        didRenderHydrated = true;
        floorBody.dataset.hydrated = '1';
        processContentForSentenceIds(floorBody);
      }
    }
  }

  // ✅ 关键：占位渲染阶段不要 init 背景（否则扫不到句子节点就“清空背景”）
  if (isBgEffectsDisabled()) return;
  if (!didRenderHydrated) return;

  if (typeof initReadingBackgroundSystem === 'function') {
    initReadingBackgroundSystem();
  } else {
    if (typeof recalcBgSegmentPositions === 'function') {
      if (typeof updateReadingBackgroundByScroll === 'function') {
        recalcBgSegmentPositions();
        updateReadingBackgroundByScroll(false);
      }
    }
  }
}

let shiftDown = false;

function shouldIgnoreKeyTarget(target) {
  if (!target) return false;
  if (target.isContentEditable) return true;

  var tag = (target.tagName || '').toLowerCase();
  if (tag === 'input') return true;
  if (tag === 'textarea') return true;
  if (tag === 'select') return true;
  return false;
}

    if (shiftDown) return;
    if (shouldIgnoreKeyTarget(e.target)) return;
    shiftDown = true;
    shiftUsedWithOther = false;
    return;
  }

  if (shiftDown) {
    shiftUsedWithOther = true;
  }
});

        toggleBgPeekMode();
      }
    }
  }

  shiftDown = false;
  shiftUsedWithOther = false;
});

async function renderScrollMode(initialTargetPageIndex = 0) {
  currentMode = 'scroll';
  if (!bookContent) return;

  bookContent.innerHTML = '';
  renderedFloorCount = 0;

  const total = Array.isArray(activePages) ? activePages.length : 0;
  if (!total) return;

  // ✅ 关键：进入 scroll 模式时，重置“按需内容准备”游标（不要全量预处理）
  resetLazyPrepareState();

  // 弱网模式：强制从第 1 楼开始，不跳转到目标楼层
  const safeTarget = isWeakLazyMode
    ? 0
    : Math.min(Math.max(initialTargetPageIndex, 0), total - 1);

  const initialEnd = isWeakLazyMode
    ? Math.min(total, WEAK_INITIAL_PAGES)
    : Math.min(total, Math.max(FLOOR_BATCH_SIZE, safeTarget + 1));

  // 先渲染楼层壳（floor-body 会是占位“正在加载...”）
  renderFloorsRange(0, initialEnd);
  renderedFloorCount = initialEnd;

  // 侧边栏只负责总楼层导航
  renderSidebar(total, 'floor');

  // 绑定懒加载滚动
  setupFloorLazyLoading();

  // ✅ 显示加载提示 + 启动超时监测
  showLazyToast('正在加载…');
  startLazySlowWatchdog();

  try {
    // ✅ 按需准备首批楼层内容（用户进入页面：算 userInitiated）
    await ensurePagesPreparedUpTo(initialEnd, { userInitiated: true });

    // ✅ hydrate：把占位替换为真实内容
    hydrateRenderedFloorsRange(0, initialEnd);

    // ✅ 把评论数（…）更新成真实数字
    for (let i = 0; i < initialEnd; i++) {
      const btn = document.querySelector(`.floor[data-page-index="${i}"] .comment-view-btn`);
      if (btn) btn.textContent = `查看本楼评论（${getCommentsCountForPage(i)}）`;
    }
  } finally {
    // 弱网模式提示一般要常驻（你也可以改成延时隐藏）
    if (!isWeakLazyMode) hideLazyToast();
  }

  // ✅ 非弱网模式下才滚动到目标楼层
  if (!isWeakLazyMode) {
  if (safeTarget > 0) {
    const targetFloor = document.querySelector(`.floor[data-page-index="${safeTarget}"]`);
    if (targetFloor) {
      boostBgHeartbeat(6000);
      targetFloor.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  }
}


  // ✅ 记录位置：弱网模式一定记 0（第 1 楼）
  saveLastViewToStorage('scroll', safeTarget);
}

// 确保某一楼已经被渲染（供导航/外部跳转使用）
function ensureFloorRendered(targetPageIndex) {
    const total = activePages ? activePages.length : 0;
    if (!total) return;

    if (targetPageIndex < renderedFloorCount) return;

    const needEnd = Math.min(
        total,
        Math.max(renderedFloorCount, targetPageIndex + 1 + FLOOR_BATCH_SIZE)
    );

    if (needEnd > renderedFloorCount) {
        renderFloorsRange(renderedFloorCount, needEnd);
        renderedFloorCount = needEnd;
    }
}

function setupFloorLazyLoading() {
  if (floorLazyScrollHandler) {
    window.removeEventListener('scroll', floorLazyScrollHandler);
  }

  let lastTriggerAt = 0;

  floorLazyScrollHandler = () => {
    if (currentMode !== 'scroll') return;

    // ⭐ 1) 记录最近阅读楼层（保留你原逻辑）
    if (bookContent) {
      const floorEls = bookContent.querySelectorAll('.floor');
      if (floorEls.length > 0) {
        const viewportTop = window.scrollY;
        const viewportMiddle = viewportTop + window.innerHeight * 0.3;

        let nearestIndex = 0;
        let minDiff = Infinity;

        floorEls.forEach(el => {
          const rect = el.getBoundingClientRect();
          const floorTop = rect.top + window.scrollY;
          const diff = Math.abs(floorTop - viewportMiddle);
          if (diff < minDiff) {
            minDiff = diff;
            const idxStr = el.getAttribute('data-page-index') || '0';
            const idx = parseInt(idxStr, 10);
            if (!isNaN(idx)) nearestIndex = idx;
          }
        });

        saveLastViewToStorage('scroll', nearestIndex);
      }
    }

    // ⭐ 2) 触底懒加载（改为 requestLoadMoreFloors）
    const scrollPos = window.scrollY + window.innerHeight;
    const docHeight = document.documentElement.scrollHeight || document.body.scrollHeight;

    // 防抖：避免滚动回弹/多次触发
    if (scrollPos + 300 >= docHeight) {
      const now = Date.now();
      if (now - lastTriggerAt < 350) return;
      lastTriggerAt = now;

      requestLoadMoreFloors();
    }

    // ⭐ 3) 弱网模式：当只渲染了前两楼时，用户继续滑到底也会继续加载（requestLoadMoreFloors 已覆盖）
    // 不需要额外分支
  };

  // 建议 passive，减少滚动开销
  window.addEventListener('scroll', floorLazyScrollHandler, { passive: true });
}


async function requestLoadMoreFloors() {
  if (isLoadingMoreFloors) return;
  if (currentMode !== 'scroll') return;
  if (!Array.isArray(activePages)) return;

  const total = activePages.length;
  if (renderedFloorCount >= total) return;

  isLoadingMoreFloors = true;

  const prevStart = renderedFloorCount;

  // ✅ 关键改动：弱网每次 +2（懒懒加载），正常每次 +FLOOR_BATCH_SIZE（比如 5）
  const step = isWeakLazyMode ? WEAK_INITIAL_PAGES : FLOOR_BATCH_SIZE;
  const nextEnd = Math.min(total, renderedFloorCount + step);

  // 防御：万一 step 配置异常导致 nextEnd 不动，避免死循环
  if (nextEnd <= prevStart) {
    isLoadingMoreFloors = false;
    return;
  }

  // 先渲染楼层壳（floor-body 用占位“正在加载...”）
  showLazyToast('正在加载…');
  renderFloorsRange(prevStart, nextEnd);
  renderedFloorCount = nextEnd;

  // 启动 60s 监测：准备太久就切弱网模式
  startLazySlowWatchdog();

  try {
    // 关键：按需准备内容（用户触发：弱网模式也允许继续）
    await ensurePagesPreparedUpTo(nextEnd, { userInitiated: true });

    // 把占位替换为真实 segmentedHtml
    hydrateRenderedFloorsRange(prevStart, nextEnd);

    // 更新评论数按钮（把 … 替换成数字）
    for (let i = prevStart; i < nextEnd; i++) {
      const btn = document.querySelector(`.floor[data-page-index="${i}"] .comment-view-btn`);
      if (btn) btn.textContent = `查看本楼评论（${getCommentsCountForPage(i)}）`;
    }
  } finally {
    isLoadingMoreFloors = false;
    // 弱网模式下你可能希望提示常驻，所以这里仅在非弱网时隐藏
    if (!isWeakLazyMode) hideLazyToast();
  }
}

        // 初始化 / 重建 背景区间列表（起始句 / 结束句 -> 垂直位置）
function initReadingBackgroundSystem() {
  // 1️⃣ 作品是否存在
  if (!activeWork) {
    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 没有 activeWork，跳过背景系统');
    }
    return;
  }

  // 2️⃣ 背景 DOM 是否就绪
  if (!bgWrapper || !bgLayerA || !bgLayerB) {
    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 背景 DOM 不完整，wrapper / layerA / layerB 有缺失：', {
        bgWrapper: !!bgWrapper,
        bgLayerA: !!bgLayerA,
        bgLayerB: !!bgLayerB
      });
    }
    return;
  }

  // 3️⃣ 读取作品里的背景配置
  const conf = pickBackgroundConfigForRead(activeWork);
  const images = Array.isArray(conf.images) ? conf.images : [];
  const bindings = Array.isArray(conf.bindings) ? conf.bindings : [];

   // ⭐ 新增：把当前配置记到全局，给 pickBgTransitionEffect 用
  bgConfig = {
    images,
    bindings,
    transitions: Array.isArray(conf.transitions) ? conf.transitions : []
  };

  if (EFFECT_BG_DEBUG) {
    console.log('🧩 [BG] 原始 background 配置:', conf);
    console.log('🧩 [BG] images:', images);
    console.log('🧩 [BG] bindings:', bindings);
  }

  bgSegments = [];
  const imageMap = {};

  // 4️⃣ 建立 imageId -> 图片信息 的映射
  images.forEach(function (img, idx) {
    if (!img) {
      return;
    }

    var key = img.imageId || img.id;
    if (!key) {
      key = 'bg-' + String(idx);
    }

    imageMap[String(key)] = {
      id: String(key),
      url: img.url,
      name: img.name || ''
    };
  });

  if (EFFECT_BG_DEBUG) {
    console.log('🧩 [BG] imageMap:', imageMap);
  }

  // 5️⃣ 找到当前真正显示句子的容器
  const containerEl = getCurrentReadingContainer();
  if (!containerEl) {
    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 找不到阅读容器');
    }
    return;
  }

  const availableSentenceEls = Array.from(
    containerEl.querySelectorAll('[data-sentence-index]')
  ).sort(function (a, b) {
    return Number(a.dataset.sentenceIndex) - Number(b.dataset.sentenceIndex);
  });
  if (!availableSentenceEls.length) {
    clearReadingBackground();
    return;
  }
  const minAvailableLine = Number(availableSentenceEls[0].dataset.sentenceIndex);
  const maxAvailableLine = Number(
    availableSentenceEls[availableSentenceEls.length - 1].dataset.sentenceIndex
  );

  bgConfig.transitions = bgConfig.transitions.flatMap(function (transition) {
    if (!transition) return [];
    const rawFromLine = Number(transition.fromLine);
    const rawToLine = Number(transition.toLine);
    const fromEl = resolveAnchoredSentenceElement(
      containerEl,
      rawFromLine,
      transition.fromAnchor
    );
    const toEl = resolveAnchoredSentenceElement(
      containerEl,
      rawToLine,
      transition.toAnchor
    );
    const fromInside =
      rawFromLine >= minAvailableLine ? rawFromLine <= maxAvailableLine : false;
    const toInside =
      rawToLine >= minAvailableLine ? rawToLine <= maxAvailableLine : false;
    if (transition.fromAnchor) {
      if (fromInside) {
        if (!fromEl) return [];
      }
    }
    if (transition.toAnchor) {
      if (toInside) {
        if (!toEl) return [];
      }
    }

    const fromLine = fromEl ? Number(fromEl.dataset.sentenceIndex) : rawFromLine;
    const toLine = toEl ? Number(toEl.dataset.sentenceIndex) : rawToLine;
    if (!Number.isInteger(fromLine) || !Number.isInteger(toLine) || fromLine >= toLine) return [];
    return [{ ...transition, fromLine, toLine }];
  });

  if (EFFECT_BG_DEBUG) {
    let maxIdx = -1;
    availableSentenceEls.forEach(el => {
      const n = Number(el.dataset.sentenceIndex);
      if (!Number.isNaN(n) ? n > maxIdx : false) maxIdx = n;
    });
    console.log('🧩 [BG] 当前 DOM 里句子节点数量 =', availableSentenceEls.length, '，最大 index =', maxIdx);
  }

  // 6️⃣ 按绑定生成每一段背景区间
  bindings.forEach(function (b, idx) {
    if (!b) {
      return;
    }

    const startLine = Number(b.startLine);
    const endLine = Number(b.endLine);
    const img = imageMap[String(b.imageId)];

    const resolvedStartEl = resolveAnchoredSentenceElement(containerEl, startLine, b.startAnchor);
    const resolvedEndEl = resolveAnchoredSentenceElement(containerEl, endLine, b.endAnchor);
    const startInside =
      startLine >= minAvailableLine ? startLine <= maxAvailableLine : false;
    const endInside =
      endLine >= minAvailableLine ? endLine <= maxAvailableLine : false;
    if (b.startAnchor) {
      if (startInside) {
        if (!resolvedStartEl) return;
      }
    }
    if (b.endAnchor) {
      if (endInside) {
        if (!resolvedEndEl) return;
      }
    }

    const resolvedStartLine = resolvedStartEl
      ? Number(resolvedStartEl.dataset.sentenceIndex)
      : startLine;
    const resolvedEndLine = resolvedEndEl
      ? Number(resolvedEndEl.dataset.sentenceIndex)
      : endLine;
    const visibleStartLine = Math.max(resolvedStartLine, minAvailableLine);
    const visibleEndLine = Math.min(resolvedEndLine, maxAvailableLine);
    const startEl = availableSentenceEls.find(function (el) {
      return Number(el.dataset.sentenceIndex) >= visibleStartLine;
    });
    const endEl = availableSentenceEls.slice().reverse().find(function (el) {
      return Number(el.dataset.sentenceIndex) <= visibleEndLine;
    });

    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 处理绑定 #' + idx, {
        imageId: b.imageId,
        imageName: b.imageName,
        startLine: startLine,
        endLine: endLine,
        resolvedStartLine: resolvedStartLine,
        resolvedEndLine: resolvedEndLine,
        visibleStartLine: visibleStartLine,
        visibleEndLine: visibleEndLine,
        hasImg: !!img,
        hasStartEl: !!startEl,
        hasEndEl: !!endEl,
        startText: startEl ? startEl.textContent.trim().slice(0, 20) : null,
        endText: endEl ? endEl.textContent.trim().slice(0, 20) : null
      });
    }

    if (!img) {
      return;
    }
    if (visibleStartLine > visibleEndLine) {
      return;
    }
    if (!startEl) {
      return;
    }
    if (!endEl) {
      return;
    }
    if (!Number.isInteger(resolvedStartLine) || !Number.isInteger(resolvedEndLine)) {
      return;
    }
    if (resolvedStartLine > resolvedEndLine) {
      return;
    }

    bgSegments.push({
      imageId: img.id,
      imageUrl: img.url,
      imageName: b.imageName || img.name,
      startLine: resolvedStartLine,
      endLine: resolvedEndLine,
      startEl: startEl,
      endEl: endEl,
      startTop: 0,
      endBottom: 0
    });
  });

  if (EFFECT_BG_DEBUG) {
    console.log('🧩 [BG] 最终生成的 bgSegments:', bgSegments);
  }

  if (!bgSegments.length) {
    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 没有任何有效的背景区间，保持默认灰色背景');
    }
    clearReadingBackground();
    return;
  }

  // 7️⃣ 先计算一次各区间的位置，再按当前滚动位置更新一次背景
  recalcBgSegmentPositions();
  updateReadingBackgroundByScroll(true);

  // 8️⃣ 只在第一次调用时绑定 scroll / resize 事件
  if (!window.__BG_SCROLL_BOUND__) {
    window.addEventListener(
      'scroll',
      function () {
        boostBgHeartbeat(900); 
        updateReadingBackgroundByScroll(false);
      },
      { passive: true }
    );

    window.addEventListener('resize', function () {
      boostBgHeartbeat(1500); 
      recalcBgSegmentPositions();
      updateReadingBackgroundByScroll(false);
    });

    window.__BG_SCROLL_BOUND__ = true;
    startBgHeartbeat();

    if (EFFECT_BG_DEBUG) {
      console.log('🧩 [BG] 已绑定 scroll / resize 事件');
    }
  }
}

        // 重新计算每个“背景区间”的起始 / 结束 Y 坐标（绝对位置）
        function recalcBgSegmentPositions() {
            if (!bgSegments.length) return;
            const scrollY = window.scrollY || document.documentElement.scrollTop || 0;

            bgSegments.forEach(seg => {
                if (!seg.startEl || !seg.endEl) return;
                const startRect = seg.startEl.getBoundingClientRect();
                const endRect = seg.endEl.getBoundingClientRect();
                seg.startTop = startRect.top + scrollY;
                seg.endBottom = endRect.bottom + scrollY;
            });
        }

function boostBgHeartbeat(ms) {
  const dur = (typeof ms === 'number') ? ms : 3000;
  const until = Date.now() + dur;
  if (until > bgHeartbeatFastUntil) {
    bgHeartbeatFastUntil = until;
  }
}

function startBgHeartbeat() {
  if (bgHeartbeatTimer) return;
  bgHeartbeatLastScrollY = -1;
  bgHeartbeatLastDocH = -1;
  bgHeartbeatMismatchStreak = 0;

  bgHeartbeatTimer = setInterval(bgHeartbeatTick, 250);
}

function stopBgHeartbeat() {
  if (!bgHeartbeatTimer) return;
  clearInterval(bgHeartbeatTimer);
  bgHeartbeatTimer = null;
}

function bgHeartbeatTick() {
  if (!bgSegments) return;
  if (bgSegments.length === 0) return;
  if (!bgLayerA) return;
  if (!bgLayerB) return;

  const now = Date.now();
  const scrollY = window.scrollY || document.documentElement.scrollTop || 0;
  const docH = document.documentElement.scrollHeight || 0;

  // 页面静止且不在“快速纠偏窗口” → 不做事（省性能）
  if (now >= bgHeartbeatFastUntil) {
    if (scrollY === bgHeartbeatLastScrollY) {
      if (docH === bgHeartbeatLastDocH) {
        return;
      }
    }
  }
  bgHeartbeatLastScrollY = scrollY;
  bgHeartbeatLastDocH = docH;

  const centerY = scrollY + window.innerHeight * 0.4;

  // currentBgSegment 可能在重建 bgSegments 后“引用失效”
  if (currentBgSegment) {
    if (bgSegments.indexOf(currentBgSegment) === -1) {
      currentBgSegment = null;
    }
  }

  // 1) 先快速判断：当前段用“实时 rect”是否仍覆盖 centerY
  if (currentBgSegment) {
    const seg = currentBgSegment;
    if (seg.startEl) {
      if (seg.endEl) {
        const sRect = seg.startEl.getBoundingClientRect();
        const eRect = seg.endEl.getBoundingClientRect();
        const freshStart = sRect.top + scrollY;
        const freshEnd = eRect.bottom + scrollY;

        if (centerY >= freshStart) {
          if (centerY <= freshEnd) {
            if (currentBgImageId === seg.imageId) {
              bgHeartbeatMismatchStreak = 0;
              return;
            }
          }
        }
      }
    }
  }

  // 2) 找到“实时位置”下，centerY 应该属于哪一段（或最近段）
  let expectedSeg = null;
  let nearestSeg = null;
  let minDist = Infinity;

  for (let i = 0; i < bgSegments.length; i++) {
    const seg2 = bgSegments[i];
    if (!seg2) continue;
    if (!seg2.startEl) continue;
    if (!seg2.endEl) continue;

    const sRect2 = seg2.startEl.getBoundingClientRect();
    const eRect2 = seg2.endEl.getBoundingClientRect();
    const start2 = sRect2.top + scrollY;
    const end2 = eRect2.bottom + scrollY;

    if (centerY >= start2) {
      if (centerY <= end2) {
        expectedSeg = seg2;
        break;
      }
    }

    let dist = 0;
    if (centerY < start2) dist = start2 - centerY;
    else if (centerY > end2) dist = centerY - end2;
    else dist = 0;

    if (dist < minDist) {
      minDist = dist;
      nearestSeg = seg2;
    }
  }

  if (!expectedSeg) expectedSeg = nearestSeg;
  if (!expectedSeg) return;

  // 3) 连续错两次（或处于 fast 窗口）才纠偏，避免 smooth scroll 中间态抖动
  if (currentBgImageId !== expectedSeg.imageId) {
    bgHeartbeatMismatchStreak++;
  } else {
    bgHeartbeatMismatchStreak = 0;
  }

  let shouldFix = false;
  if (bgHeartbeatMismatchStreak >= 2) shouldFix = true;
  if (now < bgHeartbeatFastUntil) shouldFix = true;

  if (!shouldFix) return;

  bgHeartbeatMismatchStreak = 0;

  // ✅ 核心：重算区间坐标 + 立即按滚动位置刷新背景
  recalcBgSegmentPositions();
  updateReadingBackgroundByScroll(true);
}

      
        function handleBgScrollThrottled() {
            if (bgScrollTicking) return;
            bgScrollTicking = true;
            requestAnimationFrame(() => {
                bgScrollTicking = false;
                updateReadingBackgroundByScroll(false);
            });
        }

        // 根据当前滚动位置（视口中心）决定应该使用哪个背景
        function updateReadingBackgroundByScroll(isImmediate) {
    // 1️⃣ 没有任何背景区间，直接保持默认灰色
    if (!bgSegments.length) {
        clearReadingBackground();
        return;
    }

    const scrollY = window.scrollY || document.documentElement.scrollTop || 0;
    const vh = window.innerHeight || document.documentElement.clientHeight || 0;
    const centerY = scrollY + vh / 2;   // 视口中心作为“判定点”

    // 2️⃣ 先按正常逻辑，看看 centerY 是否落在某个区间内
    let targetSeg = null;
    for (let i = 0; i < bgSegments.length; i++) {
        const seg = bgSegments[i];
        if (seg.startTop <= centerY) {
            if (centerY <= seg.endBottom) {
                targetSeg = seg;
                break;
            }
        }
    }

    // 3️⃣ 如果没命中任何区间：尝试选择“距离最近的一段”
    if (!targetSeg) {
        let nearestSeg = null;
        let minDist = Infinity;

        for (let j = 0; j < bgSegments.length; j++) {
            const seg2 = bgSegments[j];

            let dist = 0;
            if (centerY < seg2.startTop) {
                dist = seg2.startTop - centerY;   // 在该段上方
            } else if (centerY > seg2.endBottom) {
                dist = centerY - seg2.endBottom;  // 在该段下方
            } else {
                dist = 0;
            }

            if (dist < minDist) {
                minDist = dist;
                nearestSeg = seg2;
            }
        }

        const threshold = vh * 0.5;

        // ⭐ 新增：判断是否“允许退回灰色背景”
        // 如果 bindings 总数 > 已成功生成的 bgSegments 数，说明有一部分绑定还没被懒加载到 DOM 里，
        // 这时候不要清空背景，否则就会出现你现在遇到的“提前变灰一下”的情况。
        let allowClearToDefault = true;
        if (bgConfig) {
            if (Array.isArray(bgConfig.bindings)) {
                if (bgConfig.bindings.length > bgSegments.length) {
                    allowClearToDefault = false;
                }
            }
        }

        if (nearestSeg) {
            // 是否采用最近的一段背景
            let useNearest = false;

            if (minDist < threshold) {
                useNearest = true;
            } else {
                // 即使离得比较远，只要还有“未落地的绑定”，也强行用最近的一段，
                // 避免在懒加载过程中出现莫名其妙的灰色闪一下
                if (!allowClearToDefault) {
                    useNearest = true;
                }
            }

            if (useNearest) {
                targetSeg = nearestSeg;
            } else {
                // 真正离所有背景都很远，且已经确认“所有绑定都落地”时，才退回默认灰色
                if (allowClearToDefault) {
                    if (currentBgImageId !== null) {
                        playBackgroundTransition(null, 'fade', isImmediate);
                        currentBgImageId = null;
                        currentBgSegment = null;
                    }
                }
                return;
            }
        } else {
            // 连最近的一段都没有（极端情况），同样只有在允许的前提下才清空
            if (allowClearToDefault) {
                if (currentBgImageId !== null) {
                    playBackgroundTransition(null, 'fade', isImmediate);
                    currentBgImageId = null;
                    currentBgSegment = null;
                }
            }
            return;
        }
    }

    // 4️⃣ 如果判定结果和当前背景完全一样，就不重复切换
    if (currentBgSegment) {
        if (currentBgSegment === targetSeg) {
            if (currentBgImageId === targetSeg.imageId) {
                return;
            }
        }
    }

    // 5️⃣ 根据 from / to 决定转场特效
    const effectType = pickBgTransitionEffect(currentBgSegment, targetSeg);

    currentBgSegment = targetSeg;
    currentBgImageId = targetSeg.imageId;
    playBackgroundTransition(targetSeg.imageUrl, effectType, isImmediate);
}


        // 根据 fromSeg / toSeg 和 bgTransitions 决定转场特效类型
        function pickBgTransitionEffect(fromSeg, toSeg) {
            if (!bgConfig || !Array.isArray(bgConfig.transitions)) {
                return 'fade';
            }
            if (!fromSeg || !toSeg) {
                // 从默认灰背景进入某一段，或从某一段退出到默认背景：用普通淡入淡出
                return 'fade';
            }
            const fromLine = fromSeg.endLine;
            const toLine = toSeg.startLine;
            const found = bgConfig.transitions.find(function (t) {
                if (Number(t.fromLine) !== fromLine) return false;
                return Number(t.toLine) === toLine;
            });
            return found ? (found.effectType || 'fade') : 'fade';
        }
        function clearReadingBackground() {
            if (!bgLayerA || !bgLayerB) return;
            bgLayerA.style.opacity = '0';
            bgLayerB.style.opacity = '0';
            bgLayerA.style.backgroundImage = '';
            bgLayerB.style.backgroundImage = '';
            currentBgImageId = null;
            currentBgSegment = null;
        }

        // 根据 effectType 切换背景图片
        function playBackgroundTransition(nextImageUrl, effectType, isImmediate) {
    if (!bgLayerA || !bgLayerB) return;

    if (EFFECT_BG_DEBUG) {
        console.log('🧩 [BG] playBackgroundTransition 调用', {
            nextImageUrl,
            effectType,
            isImmediate,
            currentBgLayerFlag
        });
    }

    const currentLayer = currentBgLayerFlag === 'A' ? bgLayerA : bgLayerB;
    const nextLayer = currentBgLayerFlag === 'A' ? bgLayerB : bgLayerA;

    if (!nextImageUrl) {
        currentLayer.style.opacity = '0';
        nextLayer.style.opacity = '0';
        return;
    }

    nextLayer.style.backgroundImage = `url("${nextImageUrl}")`;

    if (isImmediate) {
                nextLayer.style.transition = 'none';
                currentLayer.style.transition = 'none';

                nextLayer.style.opacity = '1';
                currentLayer.style.opacity = '0';

                requestAnimationFrame(() => {
                    nextLayer.style.transition = '';
                    currentLayer.style.transition = '';
                });

                currentBgLayerFlag = currentBgLayerFlag === 'A' ? 'B' : 'A';
      if (EFFECT_BG_DEBUG) {
            console.log('🧩 [BG] 立即切换到背景', {
                layerId: nextLayer.id,
                bg: nextLayer.style.backgroundImage,
                opacity: nextLayer.style.opacity
            });
        }
                return;
            }


    nextLayer.style.opacity = '1';
    currentLayer.style.opacity = '0';
    currentBgLayerFlag = currentBgLayerFlag === 'A' ? 'B' : 'A';

    if (EFFECT_BG_DEBUG) {
        console.log('🧩 [BG] 动画切换到背景', {
            layerId: nextLayer.id,
            bg: nextLayer.style.backgroundImage,
            opacity: nextLayer.style.opacity
        });
    }
}

    });

    navTitle.appendChild(navCloseButton);
    navPopup.appendChild(navTitle);

        cell.setAttribute('aria-label', `${type === 'floor' ? '第' : '前往第'} ${i + 1} ${type === 'floor' ? '楼' : '页'}`);
        if (i === currentPageIndex) {
            cell.classList.add('active');
            cell.setAttribute('aria-current', 'page');
        }

        // ⭐ 把点击逻辑放在真正的点击回调里，而不是 for 循环里直接执行
        cell.addEventListener('click', function () {
            // 点击后关闭浮窗
            closeNavigationPopup(false);

            if (type === 'floor') {
                // 楼层模式：先保证该楼已经渲染，再滚过去
                if (typeof ensureFloorRendered === 'function') {
                    ensureFloorRendered(i);
                }

                // 下一帧再滚动，避免刚 append 时布局没更新
                setTimeout(function () {

                        targetFloor.scrollIntoView({ behavior: 'smooth', block: 'start' });
               
                    }

                    saveLastViewToStorage('scroll', i);
                }, 0);
            } else {
                // 翻页模式：切换 currentPageIndex 并重新渲染
                currentPageIndex = i;
                currentMode = 'page';
                renderCurrentMode();
                window.scrollTo(0, 0);
              // ⭐ 翻页模式导航后记录页码
                saveLastViewToStorage('page', i);
            }
        });

        navPopup.appendChild(cell);
    }
}

function setupCommentDelegation() {
    // 1. 回复（点击 comment-item）和点赞（点击 .like-btn）的事件委托
    commentList.addEventListener('click', (e) => {
        const item = e.target.closest('.comment-item');

        if (btn) {

            e.stopPropagation();

                        method: 'POST',
                        headers: { 'Authorization': `Bearer ${token}` }
                    });

                    // 更新 UI
                    const countSpan = commentItem.querySelector('.like-count');
                    countSpan.textContent = data.likesCount;
                    btn.classList.toggle('liked', data.isLikedByCurrentUser);

            const author = item.querySelector('.comment-author')?.textContent || '匿名';
            commentTextarea.value = `@${author} `;
            commentTextarea.focus();
        }

function normalizeSentenceText(t) {
  return String(t || '').replace(/\s+/g, ' ').trim();
}
function clipText(t, maxLen = 160) {
  const s = String(t || '');
  if (s.length <= maxLen) return s;
  return s.slice(0, maxLen) + '…';
}
function safeSentenceSelector(sentenceId) {
  const sid = String(sentenceId || '');
  return `[data-sentence-id="${sid.replace(/"/g, '\\"')}"]`;
}
function getSentenceTextFromDom(sentenceId) {
  if (!sentenceId) return '';
  const el = document.querySelector(safeSentenceSelector(sentenceId));
  if (!el) return '';
  return normalizeSentenceText(el.textContent);
}
function extractSentenceIdsAndTextMapFromHtml(htmlContent) {
  const ids = [];
  const idSet = new Set();
  const textMap = new Map();

  const tempDiv = document.createElement('div');
  tempDiv.innerHTML = htmlContent || '';

  tempDiv.querySelectorAll('[data-sentence-id]').forEach(el => {
    const sid = el.getAttribute('data-sentence-id');
    if (!sid) return;

    if (!idSet.has(sid)) {
      idSet.add(sid);
      ids.push(sid);
    }

    const txt = normalizeSentenceText(el.textContent);
    if (txt ? !textMap.has(sid) : false) textMap.set(sid, txt);
  });

  return { ids, idSet, textMap };
}

/* ---------- 评论：渲染 & 操作 ---------- */
function renderComments(comments, sentenceTextMap) {
  commentList.innerHTML = '';

  if (!Array.isArray(comments) || comments.length === 0) {
    commentList.innerHTML = '<p class="no-comments-tip">暂无评论，快来发表第一条评论吧！</p>';
    return;
  }

  const hasMap = sentenceTextMap instanceof Map;

  comments.forEach(c => {
    const item = document.createElement('div');
    item.className = 'comment-item';
    item.dataset.commentId = c._id;

    const author = escapeHtml(c.author || '匿名');
    const content = escapeHtml(c.content || '');
    const likes = Number(c.likesCount || 0);
    const likedClass = c.isLikedByCurrentUser ? 'liked' : '';

    // ✅ 新增：原句框（仅在传入 map 时显示）
    let quoteHtml = '';
    if (hasMap) {
      const sid = c ? (c.sentenceId ? String(c.sentenceId) : '') : '';
      const full = sid ? (sentenceTextMap.get(sid) || '') : '';
      const finalFull = full ? full : '（原句未加载）';
      const short = clipText(finalFull, 160);

      if (sid) item.dataset.sentenceId = sid;

      quoteHtml = `
        <div class="comment-context-box" title="${escapeHtml(finalFull)}">
          <div class="comment-context-title">评论原句</div>
          <div class="comment-context-text">${escapeHtml(short)}</div>
        </div>
      `;
    }

    item.innerHTML = `
      ${quoteHtml}
      <strong class="comment-author">${author}</strong>
      <p class="comment-content">${content}</p>
      <div class="comment-actions">
        <button class="like-btn ${likedClass}">❤</button>
        <span class="like-count">${likes}</span>
      </div>
    `;

    commentList.appendChild(item);
  });
}

async function openCommentPanelForPage(pageIndex) {
  const pageObj = activePages[pageIndex];
  if (!pageObj) return;

  // 没准备好就先准备（用户触发：允许弱网模式继续加载）
  if (!pageObj.segmentedHtml) {
    showLazyToast('正在加载本楼内容…');
    await ensurePagesPreparedUpTo(pageIndex + 1, { userInitiated: true });
    hydrateRenderedFloorsRange(pageIndex, pageIndex + 1);
    hideLazyToast();
  }

  const pageTitle = `第 ${pageIndex + 1} 楼/页`;
  const commentPanel = document.getElementById('comment-panel');
  const header = commentPanel.querySelector('h4');

  disableCommentInputArea(true);
  activeSentenceEl = null;

  // ✅ 直接用已缓存 HTML
  const htmlContent = pageObj.segmentedHtml || '<p>（空白）</p>';

  // ✅ 一次解析：拿到 sentenceIds（去重+保序）+ 原句映射
  const extracted = extractSentenceIdsAndTextMapFromHtml(htmlContent);
  const pageSentenceIds = extracted.ids;     // 这是数组（保序）
  const idSet = extracted.idSet;             // 用于去重
  const sentenceTextMap = extracted.textMap; // sentenceId -> 原句

  const workId = getActiveWorkIdForComments();
  if (!workId) { alert('作品尚未加载完成'); return; }

  // ✅ 把 sentenceStartIndex/count 作为“补全”，但用 Set 去重，避免重复汇总
  if (typeof pageObj.sentenceStartIndex === 'number'
    ? typeof pageObj.__sentenceIdsCount === 'number'
    : false) {
    const start = pageObj.sentenceStartIndex;
    const count = pageObj.__sentenceIdsCount;

    for (let k = 0; k < count; k++) {
      const sid = buildSentenceId(workId, start + k);
      if (!idSet.has(sid)) {
        idSet.add(sid);
        pageSentenceIds.push(sid);
      }
    }
  }

  // 只确保本楼范围的评论已加载
  await ensureCommentsLoadedForPagesRange(pageIndex, pageIndex + 1);

  // ✅ 汇总本楼评论：再按 commentId 去重（双保险）
  const commentsForPage = [];
  const seenCommentIds = new Set();

  for (let i = 0; i < pageSentenceIds.length; i++) {
    const sid = pageSentenceIds[i];
    const arr = commentBySentenceId.get(sid);
    if (!Array.isArray(arr)) continue;

    for (let j = 0; j < arr.length; j++) {
      const c = arr[j];
      if (!c) continue;

      const cid = c._id ? String(c._id) : '';
      if (cid) {
        if (seenCommentIds.has(cid)) continue;
        seenCommentIds.add(cid);
      }

      commentsForPage.push(c);
    }
  }

  commentsForPage.sort((a, b) => {
    const ta = new Date(a.createdAt).getTime();
    const tb = new Date(b.createdAt).getTime();
    if (ta < tb) return -1;
    if (ta > tb) return 1;
    return 0;
  });

  // ✅ 只渲染一次，并把原句映射传进去
  renderComments(commentsForPage, sentenceTextMap);
  header.textContent = `${pageTitle} 的所有评论 (${commentsForPage.length} 条)`;
    closeNavigationPopup(false);
    commentPanel.classList.add('show');
    commentPanel.setAttribute('aria-hidden', 'false');
}

      
async function openCommentPanelForSentence(sentenceId, isNewCommentMode = false) {
  const panel = document.getElementById('comment-panel');
  const header = panel.querySelector('h4');

  if (!sentenceId) {
    alert('句子ID无效');
    return;
  }

  // 统一取 workId（优先用你批量方案里的函数）
  let workId = null;
  if (typeof getActiveWorkIdForComments === 'function') {
    workId = getActiveWorkIdForComments();
  } else {
    if (activeWork) {
      if (activeWork._id) workId = activeWork._id;
      else if (activeWork.workId) workId = activeWork.workId;
    }
  }

  if (!workId) {
    alert('作品尚未加载完成');
    return;
  }

  // 先显示加载态
  header.textContent = '正在加载评论…';
  renderComments([]);

  // ✅ 关键：只加载该句的评论（不再全量拉取）
  try {
    await ensureCommentsLoadedForSentenceId(sentenceId);
  } catch (e) {
    console.error('加载句子评论失败:', e);
    header.textContent = '加载评论失败';
    renderComments([]);
    panel.classList.add('show');
    return;
  }

  const arr = commentBySentenceId.get(sentenceId);
  const filteredComments = Array.isArray(arr) ? arr : [];
  renderComments(filteredComments);

  if (isNewCommentMode) {
    header.textContent = `评论目标句子（已有 ${filteredComments.length} 条）`;
    disableCommentInputArea(false);

    if (!activeSentenceEl) {
      activeSentenceEl = document.querySelector(`[data-sentence-id="${sentenceId}"]`);
    } else {
      if (activeSentenceEl.dataset) {
        if (activeSentenceEl.dataset.sentenceId !== sentenceId) {
          activeSentenceEl = document.querySelector(`[data-sentence-id="${sentenceId}"]`);
        }
      } else {
        activeSentenceEl = document.querySelector(`[data-sentence-id="${sentenceId}"]`);
      }
    }

    currentViewingSentenceId = sentenceId;
    setTimeout(() => commentTextarea?.focus?.(), 0);
  } else {
    header.textContent = `句子评论 (${filteredComments.length} 条)`;
    disableCommentInputArea(true);
    currentViewingSentenceId = sentenceId;
    activeSentenceEl = null;
  }

  // 清理旧高亮
  const oldTargets = document.querySelectorAll('.active-comment-target');
  oldTargets.forEach(el => el.classList.remove('active-comment-target'));

  // 标记当前句子
  const targetEl = document.querySelector(`[data-sentence-id="${sentenceId}"]`);
  if (targetEl) targetEl.classList.add('active-comment-target');

  closeNavigationPopup(false);
  panel.classList.add('show');
  panel.setAttribute('aria-hidden', 'false');
}

async function submitComment() {
  const token = localStorage.getItem('token');
  if (!token) { alert('请先登录才能发表评论。'); return; }

  if (!activeWork) { alert('作品尚未加载完成。'); return; }
  if (!activeSentenceEl) { alert('评论目标未选中。'); return; }

  const workIdRaw = activeWork._id || activeWork.workId;
  if (!workIdRaw) { alert('作品ID缺失。'); return; }

  const sentenceId = activeSentenceEl.dataset ? activeSentenceEl.dataset.sentenceId : null;
  if (!sentenceId) { alert('该段落无法评论。'); return; }

  const content = String(commentTextarea.value || '').trim();
  if (!content) { alert('评论内容不能为空。'); return; }

  try {
    const res = await fetch(API_COMMENTS, {
      method: 'POST',
      headers: Object.assign({ 'Content-Type': 'application/json' }, getAuthHeader()),
      body: JSON.stringify({
        workId: workIdRaw,
        sentenceId: sentenceId,
        content: content
      })
    });

    if (!res.ok) {
      const errJson = await res.json().catch(() => null);
      const msg = errJson ? (errJson.message ? errJson.message : '发表评论失败') : '发表评论失败';
      throw new Error(msg);
    }

    // ✅ 关键：拿到新评论
    const saved = await res.json();

    alert('评论发表成功！');
    commentTextarea.value = '';

    // ====== ✅ 写入“哪些句子有评论”的集合 ======
    if (!commentedSentenceIds) commentedSentenceIds = new Set();
    if (saved ? Boolean(saved.sentenceId) : false) commentedSentenceIds.add(saved.sentenceId);
    commentedIdsReady = true;

    // ====== ✅ 写入“评论内容缓存” ======
    let workId = null;
    if (typeof getActiveWorkIdForComments === 'function') {
      workId = getActiveWorkIdForComments();
    } else {
      workId = workIdRaw;
    }

    if (workId) {
      if (typeof resetCommentCacheForWork === 'function') {
        if (typeof commentLoadedWorkId !== 'undefined') {
          if (commentLoadedWorkId !== workId) resetCommentCacheForWork(workId);
        }
      }

      const sid = saved.sentenceId;

      if (typeof commentBySentenceId !== 'undefined') {
        const old = commentBySentenceId.get(sid);
        if (Array.isArray(old)) old.push(saved);
        else commentBySentenceId.set(sid, [saved]);
      }

      if (typeof loadedCommentSentenceIds !== 'undefined') {
        loadedCommentSentenceIds.add(sid);
      }

      // ====== ✅ 刷新该句子的“可评论标记”（仅局部） ======
      const el = document.querySelector(`[data-sentence-id="${sid}"]`);
      if (el) {
        const floorBody = el.closest('.floor-body');
        if (floorBody) {
          if (typeof processContentForSentenceIds === 'function') {
            processContentForSentenceIds(floorBody);
          }
        }

        // ====== ✅ 更新所在楼层按钮计数（仅局部） ======
        const floor = el.closest('.floor');
        if (floor) {
          const s = floor.getAttribute('data-page-index');
          const p = parseInt(s || '-1', 10);
          if (!Number.isNaN(p)) {
            if (typeof recomputeCommentCountForPage === 'function') {
              recomputeCommentCountForPage(p, workId);
            }
            if (typeof updateFloorCommentButton === 'function') {
              updateFloorCommentButton(p);
            }
          }
        }
      }
    }

    // ✅ 重新打开该句子的评论面板（保持撰写模式）
    await openCommentPanelForSentence(sentenceId, true);

  } catch (err) {
    console.error('submitComment error', err);
                alert('发表评论失败：' + (err ? (err.message ? err.message : err) : err));
  }
}

/* ---------- 切换/渲染当前模式 ---------- */
function renderCurrentMode() {
    if (!activeWork) return;

    // ⭐ 首次渲染时，尝试根据本地记录恢复阅读位置
    if (!hasAppliedLastView) {
        const lastView = loadLastViewFromStorage();
        if (lastView ? typeof lastView.pageIndex === 'number' : false) {
            // 优先恢复到当时的模式（page / scroll）
            if (lastView.mode === 'page') {
                currentMode = 'page';
                currentPageIndex = Math.min(
                    Math.max(lastView.pageIndex, 0),
                    (activePages?.length || 1) - 1
                );
            } else {
                currentMode = 'scroll';
            }

            hasAppliedLastView = true;

            if (currentMode === 'scroll') {
                // 楼层模式：把目标楼层传给渲染函数，让它优先加载并滚动过去
                renderScrollMode(lastView.pageIndex);
            } else {
                renderPageMode();
            }

            setTimeout(() => {
                const root = currentMode === 'scroll'
                    ? bookContent
                    : (pageQuill?.root || bookContent);
                if (root) {
                    processContentForSentenceIds(root);
                    console.log('✅ 评论句子点击绑定完成（恢复阅读位置）');
                }
            }, 50);
            return; // 已按记录渲染，不再走下面的默认逻辑
        }

        // 没有记录，也标记已处理，之后切换模式不用再检查
        hasAppliedLastView = true;
    }

    // 默认：按当前模式渲染
    if (currentMode === 'scroll') {
        renderScrollMode();
    } else {
        renderPageMode();
    }

    setTimeout(() => {
        const root = currentMode === 'scroll'
            ? bookContent
            : (pageQuill?.root || bookContent);
        if (root) {
            processContentForSentenceIds(root);
            console.log('✅ 评论句子点击绑定完成');
        }
    }, 50);
}
